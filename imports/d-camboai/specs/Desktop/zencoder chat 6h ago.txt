+def compute_sma(df: pd.DataFrame, period: int, price_col: str = "close") -> pd.Series:
+    if ta:
+        return ta.SMA(df[price_col].astype(float), timeperiod=period)
+    return df[price_col].rolling(period).mean()

+def compute_ema(df: pd.DataFrame, period: int, price_col: str = "close") -> pd.Series:
+    if ta:
+        return ta.EMA(df[price_col].astype(float), timeperiod=period)
+    return df[price_col].ewm(span=period, adjust=False).mean()

+def compute_rsi(df: pd.DataFrame, period: int = 14, price_col: str = "close") -> pd.Series:
+    if ta:
+        return ta.RSI(df[price_col].astype(float), timeperiod=period)
+    # Fallback RSI
+    delta = df[price_col].diff()
+    up = np.where(delta > 0, delta, 0.0)
+    down = np.where(delta < 0, -delta, 0.0)
+    roll_up = pd.Series(up, index=df.index).ewm(alpha=1/period, adjust=False).mean()
+    roll_down = pd.Series(down, index=df.index).ewm(alpha=1/period, adjust=False).mean()
+    rs = roll_up / (roll_down + 1e-9)
+    return 100 - (100 / (1 + rs))
+

+def compute_macd(df: pd.DataFrame, fast: int = 12, slow: int = 26, signal: int = 9,
+                 price_col: str = "close") -> Tuple[pd.Series, pd.Series, pd.Series]:
+    if ta:
+        macd, macdsignal, macdhist = ta.MACD(df[price_col].astype(float),
+                                             fastperiod=fast, slowperiod=slow, signalperiod=signal)
+        return macd, macdsignal, macdhist
+    ema_fast = compute_ema(df, fast, price_col)
+    ema_slow = compute_ema(df, slow, price_col)
+    macd = ema_fast - ema_slow
+    macd_signal = macd.ewm(span=signal, adjust=False).mean()
+    macd_hist = macd - macd_signal
+    return macd, macd_signal, macd_hist

+"""
+Sentiment service using Transformers (FinBERT) to score text and provide

+try:
+    from transformers import AutoTokenizer, AutoModelForSequenceClassification
+    import torch
+except Exception:  # When transformers/torch are unavailable
+    AutoTokenizer = None
+    AutoModelForSequenceClassification = None
+    torch = None

+_LABELS = ["negative", "neutral", "positive"]
+_model_cache = {
+    "tokenizer": None,
+    "model": None,
+}
+
+
+def _load_finbert():
+    if _model_cache["tokenizer"] is not None:
+        return _model_cache["tokenizer"], _model_cache["model"]
+    if AutoTokenizer is None:
+        return None, None
+    tok = AutoTokenizer.from_pretrained("ProsusAI/finbert")
+    mdl = AutoModelForSequenceClassification.from_pretrained("ProsusAI/finbert")
+    _model_cache["tokenizer"], _model_cache["model"] = tok, mdl
+    return tok, mdl
+
+
+def finbert_score(text: str) -> Dict[str, float] | Dict[str, str]:
+    """Return {label, confidence} or fallback neutral if model missing.
+    """
+    tok, mdl = _load_finbert()
+    if tok is None or mdl is None or torch is None:
+        return {"label": "neutral", "confidence": 0.33}
+    inputs = tok(text, return_tensors="pt", truncation=True, padding=True)
+    with torch.no_grad():
+        outputs = mdl(**inputs)
+    scores = torch.nn.functional.softmax(outputs.logits, dim=1)[0]
+    idx = int(scores.argmax())
+    return {"label": _LABELS[idx], "confidence": float(scores[idx])}
+

+def zones_for_headlines(headlines: List[Dict]) -> List[Dict]:
+    """Map [{date, headline}] -> plotting windows with tone color.
+    """
+    color_map = {
+        "positive": "rgba(0,200,0,0.2)",
+        "neutral": "rgba(180,180,0,0.2)",
+        "negative": "rgba(200,0,0,0.2)",
+    }
+    zones = []
+    for item in headlines:
+        tone = finbert_score(item.get("headline", ""))
+        label = tone.get("label", "neutral") if isinstance(tone, dict) else "neutral"
+        zones.append({
+            "start": item["date"] - timedelta(days=1),
+            "end": item["date"] + timedelta(days=1),
+            "color": color_map.get(label, "rgba(180,180,0,0.2)"),
+            "label": label,
+        })
+    return zones

+def sanitize(text: str) -> Tuple[str, Dict[str, bool]]:

+                flags["phone"] = True
+            elif token == "[REDACTED URL]":
+                flags["url"] = True
+            elif token == "[REDACTED ADDRESS]":
+                flags["address"] = True
+    return out, flags

+def _openrouter_generate(system: str, messages: List[Dict[str, str]], model: Optional[str] = None) -> str:
+    key = os.getenv("OPENROUTER_API_KEY")
+    if not key:
+        return ""
+    headers = {
+        "Authorization": f"Bearer {key}",
+        "Content-Type": "application/json",
