# --- Voice Routing ---
def interpret_command(command_text):
    command_text = command_text.lower()
    if "pattern" in command_text:
        return get_chart_patterns() | get_candlestick_patterns()
    elif "indicator" in command_text:
        return get_all_indicators()
    elif "breadth" in command_text:
        return get_breadth_metrics()
    elif "macro" in command_text or "fed" in command_text:
        return get_macro_events()
    elif "volatility" in command_text or "vix" in command_text:
        return get_volatility_metrics()
    elif "institution" in command_text or "smart money" in command_text:
        return get_institutional_tactics()
    elif "market maker" in command_text:
        return get_market_maker_tactics()
    elif "execution" in command_text or "order flow" in command_text:
        return get_execution_mechanics()
    elif "style" in command_text or "scalp" in command_text:
        return get_trading_styles()
    elif "intermarket" in command_text or "correlation" in command_text:
        return get_intermarket_matrix()
    elif "psychology" in command_text or "discipline" in command_text:
        return get_trader_psychology()
    elif "journal" in command_text or "review" in command_text:
        return get_journaling_prompts()
    elif "performance" in command_text:
        return get_performance_drills()
    elif "sentiment" in command_text:
        return get_sentiment_metrics()
    elif "economic" in command_text or "calendar" in command_text:
        return get_economic_indicators()
    elif "generate strategy" in command_text:
        return generate_strategy_from_prompt(command_text)
    elif "templates" in command_text:
        return get_strategy_templates()
    else:
        return {"message": "Command not recognized. Try asking about patterns, indicators, macro events, psychology, or execution."}

# --- UI Helpers ---
def get_all_topics_by_level(level="beginner"):
    return {
        "Patterns": get_chart_patterns() if level == "beginner" else get_advanced_patterns(),
        "Indicators": get_all_indicators(),
        "Breadth": get_breadth_metrics(),
        "Macro": get_macro_events(),
        "Volatility": get_volatility_metrics(),
        "Institutional": get_institutional_tactics(),
        "Market Makers": get_market_maker_tactics(),
        "Execution": get_execution_mechanics(),
        "Styles": get_trading_styles(),
        "Intermarket": get_intermarket_matrix(),
        "Psychology": get_trader_psychology(),
        "Sentiment": get_sentiment_metrics(),
        "Economic Data": get_economic_indicators()
    }
